-- =====================================================
-- 2025教育库
CREATE DATABASE IF NOT EXISTS education_2025;
USE education_2025;
CREATE TABLE semester (
    semester_id       INT AUTO_INCREMENT PRIMARY KEY,
    academic_year     INT NOT NULL,
    semester_code     VARCHAR(20) NOT NULL,
    start_date_key    INT NOT NULL,
    end_date_key      INT NOT NULL
);

CREATE TABLE department (
    department_id     INT AUTO_INCREMENT PRIMARY KEY,
    department_code   VARCHAR(20) NOT NULL,
    department_name   VARCHAR(100) NOT NULL
);

CREATE TABLE teacher (
    teacher_id        INT AUTO_INCREMENT PRIMARY KEY,
    teacher_no        VARCHAR(20) NOT NULL,
    teacher_name      VARCHAR(100) NOT NULL,
    department_id     INT NOT NULL,
    gender_code       CHAR(1),
    phone             VARCHAR(20),
    region_id         INT,
    FOREIGN KEY (department_id) REFERENCES department(department_id)
);

CREATE TABLE class (
    class_id          INT AUTO_INCREMENT PRIMARY KEY,
    class_code        VARCHAR(20) NOT NULL,
    class_name        VARCHAR(100) NOT NULL,
    department_id     INT NOT NULL,
    grade_year        INT,
    FOREIGN KEY (department_id) REFERENCES department(department_id)
);

CREATE TABLE student (
    student_id            INT AUTO_INCREMENT PRIMARY KEY,
    student_no            VARCHAR(20) NOT NULL,
    student_name          VARCHAR(100) NOT NULL,
    gender_code           CHAR(1),
    academic_level_code   VARCHAR(20),
    department_id         INT NOT NULL,
    class_id              INT NOT NULL,
    home_region_id        INT,
    student_status_code   VARCHAR(20),
    admit_date_key        INT,
    FOREIGN KEY (department_id) REFERENCES department(department_id),
    FOREIGN KEY (class_id) REFERENCES class(class_id)
);

CREATE TABLE course (
    course_id         INT AUTO_INCREMENT PRIMARY KEY,
    course_code       VARCHAR(20) NOT NULL,
    course_name       VARCHAR(200) NOT NULL,
    department_id     INT NOT NULL,
    teacher_id        INT NOT NULL,
    credit            DECIMAL(3,1),
    course_type       VARCHAR(20),
    max_score         INT,
    FOREIGN KEY (department_id) REFERENCES department(department_id),
    FOREIGN KEY (teacher_id) REFERENCES teacher(teacher_id)
);

CREATE TABLE enrollment (
    enrollment_id     INT AUTO_INCREMENT PRIMARY KEY,
    student_id        INT NOT NULL,
    course_id         INT NOT NULL,
    semester_id       INT NOT NULL,
    enroll_date_key   INT,
    take_mode         VARCHAR(20),
    is_withdraw       TINYINT(1),
    FOREIGN KEY (student_id) REFERENCES student(student_id),
    FOREIGN KEY (course_id) REFERENCES course(course_id),
    FOREIGN KEY (semester_id) REFERENCES semester(semester_id)
);

CREATE TABLE exam_score (
    exam_score_id     INT AUTO_INCREMENT PRIMARY KEY,
    enrollment_id     INT NOT NULL,
    exam_item         VARCHAR(20),
    score             DECIMAL(5,2),
    score_date_key    INT,
    FOREIGN KEY (enrollment_id) REFERENCES enrollment(enrollment_id)
);
